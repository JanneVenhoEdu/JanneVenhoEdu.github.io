Tehtävä 1

Laske ympyrän kehän pituus ja pinta-ala

a)

\(\mathsf{r = \text{4,3 m}}\)

\(\begin{align}\mathsf{p} &= \mathsf{2 \cdot \pi \cdot r}\\ &= \mathsf{2 \cdot \pi \cdot \text{4,3 m}}\\ &= \mathsf{\text{27,01... m}}\\ &\approx \mathsf{\text{27 m}}\end{align}\)

\(\begin{align}\mathsf{A} &= \mathsf{\pi \cdot r^2}\\ &= \mathsf{\pi \cdot (\text{4,3 m})^2}\\ &= \mathsf{\text{58,08... m}^2}\\ &\approx \mathsf{\text{58 m}^2}\end{align}\)

Tehtävä 2

Laske sektorin kaaren pituus ja pinta-ala.

a)

\(\mathsf{r = \text{2,7 km}}\)

\(\mathsf{\alpha = 127^\circ}\)

\(\begin{align}\mathsf{b} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot 2 \cdot \pi \cdot r}\\ &= \mathsf{\dfrac{127^\circ}{360^\circ} \cdot 2 \cdot \pi \cdot \text{2,7 km}}\\ &= \textsf{5,984... km}\\ &\approx \textsf{6,0 km}\end{align}\)

\(\begin{align}\mathsf{A_s} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot \pi \cdot r^2}\\ &= \mathsf{\dfrac{127^\circ}{360^\circ} \cdot \pi \cdot (\text{2,7 km})^2}\\ &= \mathsf{\text{8,079... km}^2}\\ &\approx \mathsf{\text{8,1 km}^2}\end{align}\)

b)

\(\mathsf{r = \text{54,9 cm}}\)

\(\mathsf{\alpha = 90^\circ}\)

\(\begin{align}\mathsf{b} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot 2 \cdot \pi \cdot r}\\ &= \mathsf{\dfrac{90^\circ}{360^\circ} \cdot 2 \cdot \pi \cdot \text{54,9 cm}}\\ &= \textsf{86,23... cm}\\ &\approx \textsf{86,2 cm}\end{align}\)

\(\begin{align}\mathsf{A_s} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot \pi \cdot r^2}\\ &= \mathsf{\dfrac{90^\circ}{360^\circ} \cdot \pi \cdot (\text{54,9 cm})^2}\\ &= \mathsf{\text{2367,1... cm}^2}\\ &\approx \mathsf{\text{2370 cm}^2}\end{align}\)

c)

\(\mathsf{r = \text{1,47 cm}}\)

\(\mathsf{\alpha = 254^\circ}\)

\(\begin{align}\mathsf{b} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot 2 \cdot \pi \cdot r}\\ &= \mathsf{\dfrac{254^\circ}{360^\circ} \cdot 2 \cdot \pi \cdot \text{1,47 cm}}\\ &= \textsf{6,516... cm}\\ &\approx \textsf{6,52 cm}\end{align}\)

\(\begin{align}\mathsf{A_s} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot \pi \cdot r^2}\\ &= \mathsf{\dfrac{254^\circ}{360^\circ} \cdot \pi \cdot (\text{1,47 cm})^2}\\ &= \mathsf{\text{4,789... cm}^2}\\ &\approx \mathsf{\text{4,79 cm}^2}\end{align}\)

d)

\(\mathsf{d = \text{0,6 m}}\)

\(\mathsf{r = \text{0,6 m}:2 = \text{0,3 m}}\)

\(\mathsf{\alpha = 180^\circ}\)

\(\begin{align}\mathsf{b} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot 2 \cdot \pi \cdot r}\\ &= \mathsf{\dfrac{180^\circ}{360^\circ} \cdot 2 \cdot \pi \cdot \text{0,3 m}}\\ &= \textsf{0,94... m}\\ &\approx \textsf{0,9 m}\end{align}\)

\(\begin{align}\mathsf{A_s} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot \pi \cdot r^2}\\ &= \mathsf{\dfrac{180^\circ}{360^\circ} \cdot \pi \cdot (\text{0,3 m})^2}\\ &= \mathsf{\text{0,14... m}^2}\\ &\approx \mathsf{\text{0,1 m}^2}\end{align}\)

e)

\(\mathsf{r = \text{6,2 m}}\)

\(\mathsf{\alpha = 270^\circ}\)

\(\begin{align}\mathsf{b} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot 2 \cdot \pi \cdot r}\\ &= \mathsf{\dfrac{270^\circ}{360^\circ} \cdot 2 \cdot \pi \cdot \text{6,2 m}}\\ &= \textsf{29,21.. m}\\ &\approx \textsf{29 m}\end{align}\)

\(\begin{align}\mathsf{A_s} &= \mathsf{\dfrac{\alpha}{360^\circ} \cdot \pi \cdot r^2}\\ &= \mathsf{\dfrac{270^\circ}{360^\circ} \cdot \pi \cdot (\text{6,2 m})^2}\\ &= \mathsf{\text{90,57... m}^2}\\ &\approx \mathsf{\text{91 m}^2}\end{align}\)
